cmake_minimum_required(VERSION 3.18)
project(realsense_benchmark LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(RS_BENCH_ENABLE_CUDA "Build CUDA RGB preprocessing backend" ON)
option(RS_BENCH_ENABLE_LIBREALSENSE "Build D455 capture/playback tools" ON)

add_library(realsense_benchmark_core src/dataset.cpp src/cpu_backend.cpp)
target_include_directories(realsense_benchmark_core PUBLIC include)
target_compile_options(realsense_benchmark_core PRIVATE -Wall -Wextra -Wpedantic)

include(CheckLanguage)
if(RS_BENCH_ENABLE_CUDA)
  check_language(CUDA)
endif()
if(RS_BENCH_ENABLE_CUDA AND CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_library(realsense_benchmark_gpu src/cuda_backend.cu)
  target_include_directories(realsense_benchmark_gpu PUBLIC include)
  set_target_properties(realsense_benchmark_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(realsense_benchmark_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
else()
  add_library(realsense_benchmark_gpu src/cuda_backend_stub.cpp)
  target_include_directories(realsense_benchmark_gpu PUBLIC include)
endif()

add_executable(rgb_bench src/main.cpp)
target_link_libraries(rgb_bench PRIVATE realsense_benchmark_core realsense_benchmark_gpu)
target_compile_options(rgb_bench PRIVATE -Wall -Wextra -Wpedantic)

if(RS_BENCH_ENABLE_LIBREALSENSE)
  find_package(realsense2 QUIET)
  if(realsense2_FOUND)
    add_executable(realsense_live_bench src/live_benchmark.cpp)
    target_link_libraries(realsense_live_bench PRIVATE realsense2::realsense2)
    target_compile_options(realsense_live_bench PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(realsense_bag_to_rgbbin src/bag_convert.cpp)
    target_link_libraries(realsense_bag_to_rgbbin PRIVATE realsense_benchmark_core realsense2::realsense2)
    target_compile_options(realsense_bag_to_rgbbin PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(realsense_record_rgbbin src/live_record.cpp)
    target_link_libraries(realsense_record_rgbbin PRIVATE realsense_benchmark_core realsense2::realsense2)
    target_compile_options(realsense_record_rgbbin PRIVATE -Wall -Wextra -Wpedantic)
  else()
    message(STATUS "librealsense2 not found: D455 capture/playback tools will not be built")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(realsense_benchmark_test tests/test_cpu.cpp)
  target_link_libraries(realsense_benchmark_test PRIVATE realsense_benchmark_core)
  add_test(NAME realsense_benchmark_cpu_correctness COMMAND realsense_benchmark_test)
endif()
